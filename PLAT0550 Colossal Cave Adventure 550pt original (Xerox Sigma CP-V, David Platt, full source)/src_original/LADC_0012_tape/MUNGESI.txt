      SUBROUTINE SORT(MODE)
      INCLUDE MUNGE:C
      LOGICAL MODE
      DO 10 I=SYMCNT/2, 0, -1
      CALL HEAPIFY(I, SYMCNT, MODE)

      TOP = SYMCNT - I
      CALL SWAP(0, TOP)
      CALL HEAPIFY(0, TOP-1, MODE)

      RETURN
      END
      SUBROUTINE HEAPIFY(HEAD, TOP, MODE)
      INCLUDE MUNGE:C

      SUBROUTINE SWAP(I, K)
      INCLUDE MUNGE:C
      LOGICAL LSWAP
      IF (I .EQ. K) RETURN

      $NAME(K) = $NAME(I)
      $NAME(I) = $SWAP

      KEYS(K) = KEYS(I)
      KEYS(I) = SWAP
      LSWAP = REFIT(K)
      REFIT(K) = REFIT(I)
      REFIT(I) = LSWAP
      $SWAP = $UNREF(K)
      $UNREF(K) = $UNREF(I)
      $UNREF(I) = $SWAP
      RETURN
      END

      IMPLICIT INTEGER (A-Z)
