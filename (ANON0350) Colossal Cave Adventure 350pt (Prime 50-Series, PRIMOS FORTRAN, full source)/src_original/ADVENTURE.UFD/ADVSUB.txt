C    ADSUB - PRIME SPECIFIC SUBROUTINES FOR ADVENTURE
C
C    NOTE: HERE SOME OF THE SUBROUTINES WERE RECODED IN ORDER TO SPEED
C            EFFICIENCY; AS IN THE 'OR', 'AND', AND 'XOR' ROUTINES.
C            OTHERS WERE RECODED OUT OF NECESSITY (IOINIT, ETC.).
C
      SUBROUTINE DATIME(D,T)

C  RETURN THE CURRENT DATE AND TIME
C
C  D IS SET TO A NUMBER CORRESPONDING TO THE DAY OF THE WEEK
C  T IS SET TO THE NUMBER OF MINUTES SINCE MIDNIGHT
C

      IMPLICIT INTEGER (A-Z)
      REAL *8 DATE$A, TIME$A, TEMP
      INTEGER*2 DAY(8)

      CALL DATE$A(DAY)

      IF (DAY(1) .EQ. 'SU') D = 1
      IF (DAY(1) .EQ. 'MO') D = 2
      IF (DAY(1) .EQ. 'TU') D = 3
      IF (DAY(1) .EQ. 'WE') D = 4
      IF (DAY(1) .EQ. 'TH') D = 5
      IF (DAY(1) .EQ. 'FR') D = 6
      IF (DAY(1) .EQ. 'SA') D = 7

      T = TIME$A(TEMP) * 60.

      RETURN
      END



      INTEGER FUNCTION SHIFT(VAL,DIST)
      IMPLICIT INTEGER(A-Z)

C   RETURN VAL LEFT SHIFTED (LOGICALLY) DIST BITS (RIGHT SHIFT IF DIST<0)
C

      IF( DIST .GE. 0 ) SHIFT = LS( VAL, DIST )
      IF( DIST .LT. 0 ) SHIFT = RS( VAL,-DIST )

      RETURN
      END



      SUBROUTINE IOINIT(DUMMY)
$INSERT SYSCOM>KEYS.F

C   I/O SYSTEM INITIALIZATION
C
C   THIS SUBROUTINE MUST PERFORM THE FOLLOWING FUNCTIONS:
C    (1)  SET TTYI AND TTYO AS LOGICAL UNIT NUMBERS FOR THE USER'S TERMINAL
C    (2)  SET DBFI AS THE LOGICAL UNIT NUMBER OF THE DATABASE FILE
C    (3)  OPEN THE DATABASE FILE FOR INPUT ON LOGICAL UNIT DBFI
C

      IMPLICIT INTEGER(A-Z)
      INTEGER *2 X, CODE, K1, K6
      PARAMETER K1=1, K6=6
      LOGICAL BLKLIN
      COMMON /IOSCOM/ TTYI, TTYO, BLKLIN, DBFI

      TTYI = 1
      TTYO = 1
      DBFI = 5

      CALL SRCH$$(K$READ,'COMMON',K6,K1,X,CODE)
      CALL ERRPR$(K$NRTN,CODE,'IOINIT',K6,'SRCH$$',K6)

      RETURN
      END



      INTEGER FUNCTION NEWHRX(DAYTYP)

C  INPUT PRIME TIME SPECS AND SET UP A WORD OF INTERNAL FORMAT.

      IMPLICIT INTEGER(A-Z)
      LOGICAL BLKLIN
      COMMON /IOSCOM/ TTYI,TTYO,BLKLIN,DBFI

      DIMENSION TYPE(10,3)
      DATA TYPE
     1     /1HM,1HO,1HN,1H ,1H-,1H ,1HF,1HR,1HI,1H:,
     2      1HS,1HA,1HT,1H ,1H&,1H ,1HS,1HU,1HN,1H:,
     3      1HH,1HO,1HL,1HI,1HD,1HA,1HY,1HS,1H:,1H /


      NEWHRX=0
      BLKLIN=.FALSE.
      WRITE(TTYO,1)(TYPE(J,DAYTYP),J=1,10)
    1 FORMAT(' PRIME TIME ON ',10A1)

   10 WRITE(TTYO,2)
    2 FORMAT(' FROM:')
      CALL GETIN(WORD1,WORD1X,DUMMY,DUMMY,.TRUE.)
      FROM=CVSTB(WORD1,WORD1X)
      IF(FROM.LT.0.OR.FROM.GE.24)GOTO 20

      WRITE(TTYO,4)
    4 FORMAT(' TILL:')
      CALL GETIN(WORD1,WORD1X,DUMMY,DUMMY,.TRUE.)
      TILL=CVSTB(WORD1,WORD1X)
      IF(TILL.LT.FROM.OR.TILL.GE.24)GOTO 20

      DO 5 I=FROM,TILL
    5 NEWHRX=OR(NEWHRX,SHIFT(1,I))
      GOTO 10

   20 BLKLIN=.TRUE.
      RETURN
      END



      SUBROUTINE HOURSX(H,DAYTYP)

C  USED BY HOURS (ABOVE) TO PRINT HOURS FOR EITHER WEEKDAYS OR WEEKENDS.

      IMPLICIT INTEGER(A-Z)
      LOGICAL FIRST
      COMMON /IOSCOM/ TTYI,TTYO,BLKLIN,DBFI

      DIMENSION TYPE(10,3)
      DATA TYPE
     1     /1HM,1HO,1HN,1H ,1H-,1H ,1HF,1HR,1HI,1H:,
     2      1HS,1HA,1HT,1H ,1H&,1H ,1HS,1HU,1HN,1H:,
     3      1HH,1HO,1HL,1HI,1HD,1HA,1HY,1HS,1H:,1H /


      FIRST=.TRUE.
      FROM=-1
      IF(H.NE.0)GOTO 10

      WRITE(TTYO,2)(TYPE(J,DAYTYP),J=1,10)
    2 FORMAT(10X,10A1,'   OPEN ALL DAY')
      RETURN

   10 FROM=FROM+1
      IF(AND(H,SHIFT(1,FROM)).NE.0)GOTO 10
      IF(FROM.GE.24)GOTO 20
      TILL=FROM
   14 TILL=TILL+1
      IF(AND(H,SHIFT(1,TILL)).EQ.0.AND.TILL.NE.24)GOTO 14

      IF(FIRST)WRITE(TTYO,16)(TYPE(J,DAYTYP),J=1,10),FROM,TILL
      IF(.NOT.FIRST)WRITE(TTYO,18)FROM,TILL
   16 FORMAT(10X,10A1,I4,':00 TO',I3,':00')
   18 FORMAT(20X,I4,':00 TO',I3,':00')
      FIRST=.FALSE.
      FROM=TILL
      GOTO 10

   20 IF(FIRST)WRITE(TTYO,22)(TYPE(J,DAYTYP),J=1,10)
   22 FORMAT(10X,10A1,'   CLOSED ALL DAY')
      RETURN
      END



      SUBROUTINE ADDR(WORD,ADDRSS)

      END



      INTEGER FUNCTION SIZE(FIRST,LAST)
      IMPLICIT INTEGER (A-Z)

      RETURN

      END



      SUBROUTINE LDCOMN(QNAME,NAME,CMADRS,CMSZES)
$INSERT SYSCOM>KEYS.F
C
C   ADVENTURE SUBROUTINE TO LOAD ALL COMMON BLOCKS FROM EITHER
C     THE SYSTEM FILE (QNAME=.TRUE.) OR A PREVIOUSLY SAVED FILE
C     (QNAME=.FALSE.)
C
C   IF THE SYSTEM FILE IS NOT FOUND, DO  NOTHING
C
C   IF THE NAMED FILE IS NOT FOUND, DIE
C
      IMPLICIT INTEGER (A-Z)
      COMMON /IOSCOM/ TTYI,TTYO,BLKLIN,DBFI
      LOGICAL QNAME
      DIMENSION CMADRS(4,11), CMSZES(11), NAME(10)
      INTEGER *2 SYSNAM(3), K6, K1, X, CODE, I, K32,  K10
      PARAMETER K6=6, K1=1, K32=32,  K10=10
      DATA SYSNAM/'ADVCOM'/

      CALL SRCH$$(K$CLOS,0,0,K1,X,CODE)
      IF( .NOT. QNAME ) GO TO 10
      CALL SRCH$$(K$READ,SYSNAM,K6,K1,X,CODE)
      IF( CODE .NE. 0 ) RETURN
      GO TO 20

   10 CALL PACK(NAME,1,10,NAME,1)
      IF( NAME(1) .EQ. '    ' ) GO TO 40
      CALL SRCH$$(K$READ,NAME,K10,K1,X,CODE)
      CALL ERRPR$(K$NRTN,CODE,0,0,'SRCH$$',K6)

   20 DO  30  I = 1, 11
   30 CALL PRWF$$(K$READ,K1,CMADRS(1,I),INTS(CMSZES(I)),0000000,X,CODE)

      CALL SRCH$$(K$CLOS,0,0,K1,X,CODE)
      RETURN

   40 CALL TNOUA(' WHAT FILE SHOULD I SEARCH FOR ?',K32)
      READ(TTYI,50) NAME
   50 FORMAT(10 A1)
      GO TO 10
      END



      SUBROUTINE SVCOMN(QNAME,NAME,CMADRS,CMSZES)
$INSERT SYSCOM>KEYS.F
C
C   ADVENTURE SUBROUTINE TO SAVE COMMON FOR FUTURE USE
C
C   OPEN FILE ADVCOM (NAME) DEPENDING ON QNAME AND WRITE OUT ALL
C      PROGRAM VARIABLES FOR RESTART LATER
C
      IMPLICIT INTEGER (A-Z)
      COMMON /IOSCOM/ TTYI,TTYO,BLKLIN,DBFI
      LOGICAL QNAME
      DIMENSION CMADRS(4,11), CMSZES(11), NAME(10)
      INTEGER *2 SYSNAM(3), K1, K6, X, CODE, I, K10, K31
      PARAMETER K1=1, K6=6, K10=10, K31=31
      DATA SYSNAM/'ADVCOM'/

      CALL SRCH$$(K$CLOS,0,0,K1,X,CODE)
      IF( .NOT. QNAME ) GO TO 10
      CALL SRCH$$(K$WRIT,SYSNAM,K6,K1,X,CODE)
      IF( CODE .NE. 0 ) CALL ERRPR$(K$SRTN,CODE,0,0,0,0)
      GO TO 20

   10 CALL PACK(NAME,1,10,NAME,1)
      IF( NAME(1) .EQ. '    ' ) GO TO 40
      CALL SRCH$$(K$WRIT,NAME,K10,K1,X,CODE)
      IF( CODE .NE. 0 ) CALL ERRPR$(K$SRTN,CODE,0,0,0,0)

   20 CALL PRWF$$(K$TRNC,K1,0,0,000000,X,CODE)

      DO  30  I = 1, 11
   30 CALL PRWF$$(K$WRIT,K1,CMADRS(1,I),INTS(CMSZES(I)),000000,X,CODE)

      CALL SRCH$$(K$CLOS,0,0,K1,X,CODE)
      RETURN

   40 CALL TNOUA('WHAT NAME SHOULD I FILE UNDER ?',K31)
      READ(TTYI,50) NAME
   50 FORMAT(10 A1)
      GO TO 10
      END




      SUBROUTINE LEGAL
$INSERT SYSCOM>KEYS.F
$INSERT SYSCOM>ERRD.F
      INTEGER *2 BUFF(40), INFO(5), ARRAY(2), X, CODE, K3, K1, K29, K40,
     1           K5, K2
      PARAMETER K3=3, K1=1, K5=5, K29=29, K40=40

      CALL BREAK$(.TRUE.)
      CALL SRCH$$(K$RDWR,'XYZ',K3,K1,X,CODE)
      CALL SRCH$$(K$CLOS,0,0,K1,X,CODE)
      CALL ATCH$$('GAMES',K5,K$ALLD,'      ',K$IMFD,CODE)
      CALL SRCH$$(K$EXST,'XYZ',K3,K1,X,CODE)
      IF( CODE .EQ. E$FNTF ) GO TO 10
      CALL SRCH$$(K$DELE,'XYZ',K3,K1,X,CODE)
      CALL ATCH$$(K$HOME,0,0,0,0,CODE)
      CALL SRCH$$(K$DELE,'XYZ',K3,K1,X,CODE)
      CALL BREAK$(.FALSE.)
      RETURN

   10 ARRAY(2) = 0
      ARRAY(1) = :3407
      CALL ATCH$$(K$HOME,0,0,0,0,CODE)
      CALL SRCH$$(K$DELE,'XYZ',K3,K1,X,CODE)
      CALL RDTK$$(K3,INFO,BUFF,K40,CODE)
      CALL RDTK$$(K2,INFO,BUFF,K40,CODE)
      CALL RDTK$$(K2,INFO,BUFF,K40,CODE)
      CALL SATR$$(K$PROT,BUFF,INFO(2),ARRAY,CODE)
      CALL SRCH$$(K$DELE,BUFF,INFO(2),K1,X,CODE)


      CALL TNOUA('CAN''T - NOT ATTACHED TO GAMES',K29)
      CALL BREAK$(.FALSE.)
      STOP
      END



      SUBROUTINE GETTO

      SUBROUTINE PACK(IFROM,ISTRT,ISTOP,JDIM,JSTRT)
      INTEGER *2 IFROM(1), JDIM(1), JCON, ICON, I
      INTEGER *4 ISTRT, ISTOP, JSTRT

      JCON = JSTRT + ( ISTOP - ISTRT )/2
      ICON = ISTRT

      DO  10  I = JSTRT, JCON
      JDIM(I) = LT( IFROM(ICON),8 ) + RS( IFROM(ICON+2),8 )
   10 ICON = ICON + 4

      IF( MOD(ISTOP-ISTRT+1,2) .EQ. 1 )
     1      JDIM(JCON) = LT( JDIM(JCON),8 ) + :240
      RETURN
      END



      INTEGER FUNCTION RAN(RANGE)

C  RETURN RANDOM UNIFORMLY DISTRIBUTED VALUE IN CLOSED INTERVAL [0,RANGE-1].
C
C  COURTESY MICHAEL GRUBIN@PRIME COMPUTER INC., FRAMINGHAM MASS.
C

      IMPLICIT INTEGER (A-Z)
      REAL RAND$A
      COMMON /RANCOM/ R

      IF( R .EQ. 0 ) CALL RNDI$A(R)
      RAN = (RANGE-1)*RAND$A(R)
      RETURN
      END
