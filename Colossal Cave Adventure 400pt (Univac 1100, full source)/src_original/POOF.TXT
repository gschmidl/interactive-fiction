       SUBROUTINE POOF

C  AS PART OF DATABASE INITIALISATION, WE CALL POOF TO SET UP SOME DUMMY
C  PRIME-TIME SPECS, MAGIC WORDS, ETC.

       IMPLICIT INTEGER(A-Z)
      include comwiz,list               @ define WIZCOM common block

       wkday(1) = 0                                                     DK
       wkday(2) = 261888                @ (0777400 octal)               DK
       wkend(1) = 0                                                     DK
       wkend(2) = 0                                                     DK
       holid(1) = 0                                                     DK
       holid(2) = 0                                                     DK
       HBEGIN=0
       HEND=-1
       SHORT=30
       MAGIC='dwarf'
       MAGNM=11111
       LATNCY=90
       RETURN
       END
