*       title:  The King's Mission Game
*
*       author: Kent A. Wendler
*
*       © 1977 by Kent A. Wendler
*       © 1981 by the University of Illinois
*                Board of Trustees
*
*       abstract:       In the King's Mission Game, the
*               player is given a "mission" to "kill" all 
*               of a specified type of monster in one of
*               the "enchanted realms" which is selected
*               by the player and which range in size from
*               1 to 100 sectors.  Each sector is a 10×10
*               matrix of possible positions which can be
*               occupied by the player or other objects.
*               The game becomes progressively more diff-
*               icult with continued play.
*                       Game records are stored for each
*               player by means of a system which will, if
*               necessary, delete the record of the lowest
*               rated player not actively playing the game.
*               Provision is made for suspending a game for 
*               later resumption.  Various game statistics
*               are also kept and are optionally available
*               for display.  All player records are
*               cleared upon intersystem transfer.
*
*       intended audience:      ages 8 and older
*
*       prerequisites:  none
*
*       average length of time in lesson:       open-ended
*
*       machine resources:      cpu:    > 5 tips, backgnd
*                                       & foregnd options
*                               memory: 6700 words, approx.
*                               disc:   < 1 dapm
*
*       associated files:       =tkmb=  help lesson, common,
*                                       charset, microtable
*                               =tkmds= dataset for storing
*                                       suspended games
*
*       associated file references:
*               unit:   (ieu); statements 2, 12, 13, 19,
*                                         20, 63
*               unit:   page1; statement 43
*               unit:   chars; statement 9

=== block 'between'  kind 00  284 words  (block 11)
unit    intersec
*       initialize new section upon entrance
zero    secmap,10       $$ clear section map
calc    xnew«speed*cos(theta)+x $$ new position
        ynew«speed*sin(theta)+y
calc    i1«(xnew<1)-(xnew>10)   $$ find new section
        i2«(ynew<1)-(ynew>10)
calcs   i1,xsec«xsec-1,,xsec+1
calcs   i2,ysec«ysec-1,,ysec+1
if      xsec<1$or$(xsec=1$and$xnew=1)
.       calc    xsec«1
.               i8«-1   $$ border tree collision on section entry
.               xnew«2
elseif  xsec>realm$or$(xsec=realm$and$xnew=10)
.       calc    xsec«realm
.               i8«-1
.               xnew«9
else
.       calcs   i1,xnew«xnew+10,,xnew-10
.       zero    i8
endif
if      ysec<1$or$(ysec=1$and$ynew=1)
.       calc    ysec«1
.               i9«-1
.               ynew«2
elseif  ysec>realm$or$(ysec=realm$and$ynew=10)
.       calc    ysec«realm
.               i9«-1
.               ynew«9
else
.       calcs   i2,ynew«ynew+10,,ynew-10
.       zero    i9
endif
branch  xsec=1,x,1next  $$ check & set border trees
doto    1tree,i3«1,10
calc    otype(1,i3)«1
1tree   
1next   
branch  xsec=realm,x,2next
doto    2tree,i3«1,10
calc    otype(10,i3)«1
2tree   
2next   
branch  ysec=1,x,3next
doto    3tree,i3«1,10
calc    otype(i3,1)«1
3tree   
3next   
branch  ysec=realm,x,4next
doto    4tree,i3«1,10
calc    otype(i3,10)«1
4tree   
4next   secno«xsec+realm*(ysec-1)
doto    1set,i3«1,7     $$ generate map
calcs   i3-2,i4«trees(secno),treasur(secno),mcirc(secno),goblins(secno),zombies(secno),wrwolvs(secno),demons(secno)
doto    1set,i7«1,i4
loop    $$ open space search
.       randu   ix,10
.       randu   iy,10
*       inhibit placing weak player next to a demon
reloop  speed=0$and$i3=7$and$((ix-xnew)**2+(iy-ynew)**2){^72}2$and$hits<2000
outloop otype(ix,iy)=0$and$(ix{^54}xnew$or$iy{^54}ynew)
endloop 
calc    otype(ix,iy)«i3
1set
