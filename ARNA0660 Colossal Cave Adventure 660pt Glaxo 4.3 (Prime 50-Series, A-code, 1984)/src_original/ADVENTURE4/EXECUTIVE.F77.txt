      SUBROUTINE COMMAND
$INSERT EXECUTIVE.INS.F77
      CHARACTER*15 Z$HOLDARG

      LINELEN = 0
      IF (AND(BITVAL(STATUS),PLSCLRFY) .NE. 0) THEN
          HOLDARG   = LINEWORD(1)
          Z$HOLDARG = Z$ARG(1)
          ENDIF
      DO 5 I=1, LINESIZE
5     LINEWORD(I) = -1
      CALL READIN
10    CALL PARSE($100)
      CALL FIND(Z$LEX, $50, LINEVAL)
      IF (CLASS(LINEVAL) .EQ. NULL) GOTO 10
20    LINELEN = LINELEN + 1
      LINEWORD(LINELEN) = LINEVAL
      Z$ARG(LINELEN)=Z$LEX
      IF (LINELEN .LT. LINESIZE) GOTO 10
      GOTO 100
50    LINEVAL = -1
      GOTO 20
100   CONTINUE
      IF (LINELEN .EQ. 1 .AND. AND(BITVAL(STATUS),PLSCLRFY) .NE. 0) THEN
          LINEWORD(2) = HOLDARG
          Z$ARG(2)    = Z$HOLDARG
          LINELEN     = 2
          ENDIF
      CALL SETBIT(STATUS, AND(BITVAL(STATUS), NOT(PLSCLRFY)))
      I = CLASS(LINEWORD(1))
      IF (LINELEN .EQ. 2 .AND.
     +   CLASS(LINEWORD(2)) .EQ. VERB .AND.
     +   (I .EQ. OBJECT .OR. I .EQ. PLACE)) THEN
          HOLDARG     = LINEWORD(1)
          Z$HOLDARG   = Z$ARG(1)
          LINEWORD(1) = LINEWORD(2)
          Z$ARG(1)    = Z$ARG(2)
          LINEWORD(2) = HOLDARG
          Z$ARG(2)    = Z$HOLDARG
          ENDIF
      DO 99 I=1, LINESIZE
      K = 0
      J = CLASS(LINEWORD(I))
      IF (J .EQ. OBJECT .OR. J .EQ. PLACE) K = BITVAL(LINEWORD(I))
      IF (J .EQ. VERB) K = XVERB
      IF (LINEWORD(I) .LT. 0) K = BADWORD
      CALL SETVAL(ARGWORDS(I), LINEWORD(I))
99    CALL SETBIT(ARGWORDS(I), K)
      CALL SETVAL(STATUS, LINELEN)
      IF (LINELEN .EQ. 0) GOTO 999
      I = CLASS(LINEWORD(1))
      IF (I .EQ. VERB)
     + CALL SETBIT(STATUS, OR(BITVAL(STATUS), XVERB))
      IF (I .EQ. OBJECT)
     + CALL SETBIT(STATUS, OR(BITVAL(STATUS), XOBJECT))
      IF (I .EQ. PLACE)
     + CALL SETBIT(STATUS, OR(BITVAL(STATUS), XPLACE))
999   CONTINUE
      RETURN
      END
      SUBROUTINE FIND(Z$ID, *, RFIND)
$INSERT EXECUTIVE.INS.F77
      CHARACTER*(*) Z$ID
      CHARACTER*12  Z$IDE

      Z$IDE = Z$ID
      LOW = 1
      HIGH = SYMCNT

      DO 5 I=1, WORDSIZE
      IF (Z$IDE(I:I).LT.Z$LC(1:1).OR.Z$IDE(I:I).GT.Z$LC(26:26)) GOTO 5
      J = INDEX(Z$LC, Z$IDE(I:I))
      IF (J .NE. 0) Z$IDE(I:I) = Z$UC(J:J)
5     CONTINUE

10    RFIND = (LOW + HIGH) / 2
      IF (Z$IDE .EQ. Z$WORD(RFIND)) GOTO 30
      IF (LOW .GE. HIGH) Z$BADVAR=Z$IDE
      IF (LOW .GE. HIGH) RETURN 1
      IF (Z$IDE .LT. Z$WORD(RFIND)) GOTO 20
      LOW = RFIND + 1
      GOTO 10

20    HIGH = RFIND - 1
      GOTO 10

30    RFIND = RT(FILEKEY(RFIND), 15)
      RETURN
      END

      WHRE = WHERE(I)

      SUBROUTINE OPENDB
$INSERT EXECUTIVE.INS.F77

      INTEGER ZWORD
      EQUIVALENCE (Z$WORD, ZWORD)
      INTEGER LTB(2),NWR(2)
      INTEGER*4 I4,NWD,LOCTB,LNWR
      EQUIVALENCE (LTB,LOCTB),(NWR,LNWR)
C****
C* Open and read four files in the current directory: ADVINIT1.dat to
C* ADVINIT4.dat. Modify this code if the database is to be read from
C* another directory. The rigmarole with ADVINIT2 required due to the
C* file exceeding one segment.
C****
      CALL SRCH$$(K$READ,'ADVINIT1.DAT',12,DBI,TY,ERCODE)
      IF (ERCODE.NE.0) CALL ERRPR$(0,ERCODE,'ADVINIT1',8,0,0)
      CALL PRWF$$(K$READ,DBI,LOC(BUFFER),BUFFSIZE,000000,NR,ERCODE)
      IF (ERCODE.NE.1) CALL ERRPR$(0,ERCODE,'ADVINIT1',8,0,0)
      CALL SRCH$$(K$CLOS,0,0,DBI,TY,ERCODE)

      CALL SRCH$$(K$READ,'ADVINIT2.DAT',12,DBT,TY,ERCODE)
      IF (ERCODE.NE.0) CALL ERRPR$(0,ERCODE,'ADVINIT2',8,0,0)
      LNWR=0

      NWD=LNWR

      CALL PRWF$$(1,DBT,LOCTB,1,000000,NWR(2),ERCODE)
      NWD=NWD+LNWR
      LTB(1)=LTB(1)+1
      LTB(2)=0
      CALL PRWF$$(1,DBT,LOCTB,:177777,000000,NWR(2),ERCODE)
      NWD=NWD+LNWR
      IF (ERCODE.NE.1) CALL ERRPR$(0,ERCODE,'ADVINIT2',8,0,0)
      CALL SRCH$$(K$CLOS,0,0,DBT,TY,ERCODE)
      DO 10 I4=1,NWD
         TEXTBUFFER(I4)=-TEXTBUFFER(I4)

      CALL SRCH$$(K$READ,'ADVINIT3.DAT',12,DBI,TY,ERCODE)
      IF (ERCODE.NE.0) CALL ERRPR$(0,ERCODE,'ADVINIT3',8,0,0)
      CALL PRWF$$(K$READ,DBI,LOC(KEYSBF),4*KEYSIZ,000000,NR,ERCODE)
      IF (ERCODE.NE.1) CALL ERRPR$(0,ERCODE,'KEYSFILE',8,0,0)
      CALL SRCH$$(K$CLOS,0,0,DBI,0,ERCODE)

      IF (ERCODE.NE.0) CALL ERRPR$(0,ERCODE,'ADVINIT4',8,0,0)
      CALL PRWF$$(K$READ,DBI,LOC(SYMCNT),1,000000,NR,ERCODE)
      IF (ERCODE.NE.0) CALL ERRPR$(0,ERCODE,'ADVINIT4',8,0,0)
