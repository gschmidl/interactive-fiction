dungeon  : procedure options(main);

/****************************************************************/
/*   Author :  Jonathan H. Reed                                 */
/*   Date   :  6/83                                             */
/****************************************************************/

     declare (hidpoint,trap,lim,trapflag,ckfl) fixed (4);
     declare hidden (100,4) fixed (4);
     declare mem (10) character (10);
     declare memw (10,3) fixed (4);
     declare (okesp,drag) fixed (5);
     declare (mempoint,pos_sub,anger,dead,newdrag,hp) fixed (4);
     declare (old,position) (3) fixed (4);
     declare possess (50) character (10);
     declare item character (10) varying;
     declare (gold,flg1,rate,added) fixed (4);
     declare command character (8) varying;
     declare (i,u,v,x,y,z) fixed (4);
     declare (m,t) (10,10,10) fixed;
     declare (seed,rnd) float;
     declare number fixed;

     put skip list ('******** DUNGEON ********');
     put skip list ('  ');
  m1:put skip list ('Do you want instructions?');
     get skip list (item);
     if index('yes',item)=1 then call direct;
     else if index('YES',item)=1 then call direct;
     else if index('no',item)=1 then;
     else if index('NO',item)=1 then;
     else goto m1;

     put skip list ('Please input a wierd positive number.');
     call getnum;  seed=number;
     seed = floor(seed*1397) + 192;

 g1:call init;

 l1:if dead>0 then do;
     put skip list (' ');
     put skip list ('Command?');
     get skip list (command);
     call abbrev;
     if trap>0 then if command='search' then;
     else if command='checktra' then;
     else if command='help' then;
     else if command='untrap' then;
     else do; call falltrap;
        if trapflag=1 then do;
            trapflag=0;
            if anger=0 then goto l1;
            command='d';
        end;
     end;
     if command='d' then; else
     if command='move' then call move;
      else if command='attack' then call attack;
       else if command='items' then call items;
         else if command='help' then call help;
          else if command='checktra' then call checktrap;
           else if command='where' then call where;
            else if command='untrap' then call untrap;
             else if command='search' then call search;
               else if command='take' then call take;
                else if command='hide' then call hide;
                 else if command='retrieve' then call retrieve;
                   else if command='esp'  then  call lev;
                    else if command='portal' then call portal;
                     else if command='status' then call status;
                      else if command='wand' then call wand;
                       else if command='potion' then call potion;
                        else if command='quit' then;
                         else if command='remember' then call remember;
                          else if command='memory' then call memory;
                          else if command='forget' then call forget;
                          else do;
                                put skip list ('I don''t understand.');
                                goto l1;
                          end;
     end;
     if command='help' then;
     else if command='quit' then;
     else do;
         okesp=okesp-1;
         dead=dead-1;
         if anger>0 then if newdrag=0
         then call dturn;
         else newdrag=0;
     end;
     if  command =  'quit' then do;
            call score;
            put skip list ('Your final score is:',rate);
            put skip list ('                 ');
     end;
          else if dead<=0 then do;
             put skip list ('                 ');
             put skip list ('Unfortunately, this kills you!');
             put skip list ('                 ');
             put skip list ('Your',gold,'gold pieces will make a');
             put skip list ('nice addition to some dragon''s hoard!');
             put skip list ('                 '); call score;
             put skip list ('You died with a score of:',rate);
             put skip list ('                ');
          end;
          else goto l1;
  a1:put skip list ('Play again?');
     get skip list (item);
     if index('yes',item)=1 then goto g1;
     else if index('no',item)=1 then;
     else goto a1;

/* *********************************************** */
/*                procedures                       */
/* *********************************************** */

getnum:procedure;
   declare num character (4) varying;
   declare (chn,x,y) fixed;
   declare ch character (1);
l1:get skip list (num);
   y=1;
   number=0;
   x=length(num);
   do while (x>0);
      ch=substr(num,x,1);
      if ch='1' then chn=1;
      else if ch='2' then chn=2;
      else if ch='3' then chn=3;
      else if ch='4' then chn=4;
      else if ch='5' then chn=5;
      else if ch='6' then chn=6;
      else if ch='7' then chn=7;
      else if ch='8' then chn=8;
      else if ch='9' then chn=9;
      else if ch='0' then chn=0;
      else do;
        put skip list('Invalid entry, try again:');
        goto l1;
      end;
      number=number+chn*y;
      y=y*10;
      x=x-1;
   end;
end getnum;

score:procedure;
   rate=3*gold + pos_sub + 5*drag + added;
end score;

abbrev:procedure;
   declare c character (8) varying;
   c=command;
   if index('esp',c)=1 then c='esp';
   else if index('help',c)=1 then c='help';
   else if index('move',c)=1 then c='move';
   else if index('take',c)=1 then c='take';
   else if index('attack',c)=1 then c='attack';
   else if index('wand' ,c)=1 then c='wand';
   else if index('items',c)=1 then c='items';
   else if index('search',c)=1 then c='search';
   else if index('checktraps',c)=1 then c='checktra';
   else if index('hide',c)=1 then c='hide';
   else if index('remember',c)=1 then c='remember';
   else if index('potion',c)=1 then c='potion';
   else if index('portal',c)=1 then c='portal';
   else if index('status',c)=1 then c='status';
   else if index('where',c)=1 then c='where';
   else if index('memory',c)=1 then c='memory';
   else if index('forget',c)=1 then c='forget';
   else if index('retrieve',c)=1 then c='retrieve';
   else if index('untrap',c)=1 then c='untrap';
   else if index('quit',c)=1 then c='quit';
   command=c;
end abbrev;

status:procedure;
   put skip list ('Life energy level :',dead);
   if okesp<=0 then put skip list ('Esp : working');
     else put skip list ('Esp : not working');
   call score;
   put skip list ('Current score :',rate);
end status;

rand : procedure;
     declare (con,joe) float;
     declare i fixed (4);
     con=12727;
     joe =floor(seed/100);
     seed  = mod(seed+joe +con,100000);
     rnd = mod(seed,100)*.01;
end rand;

replace:procedure;
   l1:call rand;
      rnd=floor(rnd*5)+1;
      if rnd=1 then goto l1;
      if rnd=5 then rnd=7;
      m(old(1),old(2),old(3))=rnd;
      call thcord;
      if m(x,y,z)=0 then m(x,y,z)=1;
end replace;

init : procedure;
     do i=1 to 10;
     do u=1 to 10;
     do v=1 to 10;
     m(i,u,v)=0; t(i,u,v)=0;
     end;
     end;
     end;
     do i=1 to 75;      /* monsters, gold, and traps */
     call thcord;
     m(x,y,z)=1;
     do v=1 to floor(rnd*20); call rand; end;
     call thcord;
     m(x,y,z)=2;
     do v=1 to floor(rnd*20); call rand; end;
     call thcord;
     t(x,y,z)=1;
     do v=1 to floor(rnd*20); call rand; end;
     end;
     do i=1 to 50;      /* more gold and traps */
     call thcord;
     do v=1 to floor(rnd*20); call rand; end;
     m(x,y,z)=2;
     do v=1 to floor(rnd*20)+1; call rand; end;
     call thcord;
     t(x,y,z)=1;
     end;
     do i=1 to floor(rnd*20)+1; call rand; end;
     do i=1 to 12;      /* wands, potions, and monsters */
     call thcord;
     m(x,y,z)=4;
     call thcord;
     m(x,y,z)=1;
     call thcord;
     m(x,y,z)=3;
     do u=1 to floor(rnd*20)+1; call rand; end;
     end;
     do i=1 to floor(rnd*20)+1; call rand; end;
     do i=1 to 12;      /* swords and potions */
     call thcord;
     m(x,y,z)=7;
     call thcord;
     m(x,y,z)=3;
     do u=1 to floor(rnd*20)+1; call rand; end;
     end;
  l1:put skip list ('Short or long version?');
     get skip list (item);
     if index('short',item)=1 then do;
        lim=500;
        dead=500;
        v=5;
     end;
     else if index('long',item)=1 then do;
        lim=1000;
        dead=1000;
        v=3;
     end;
     else goto l1;
     do u=1 to v;
     do i=1 to floor(rnd*20)+1; call rand; end;
     call thcord;       /* fake portal */
     m(x,y,z)=5;
     do i=1 to floor(rnd*20)+1; call rand; end;
     call thcord;       /* true portal */
     m(x,y,z)=6;
     end;
     call thcord;
     position(1)=x;
     position(2)=y;
     position(3)=z;
     if t(x,y,z)=1 then trap=1; else trap=0;

 /*                1 = monster             */
 /*                2 = gold                */
 /*                3 = potion              */
 /*                4 = wand                */
 /*                5 = fake portal         */
 /*                6 = true portal         */
 /*                7 = sword               */
     ckfl=0;  flg1=0;
     added=0;
     gold=0;
     pos_sub=1;
     call where;
     anger=0;
     okesp=0;
     drag=0;
     mempoint=1;
     hidpoint=1;
     trapflag=0;
end init;

chesp:procedure;
   call rand;
   if okesp<5 then okesp=floor(rnd*15)+5;
end chesp;

thcord : procedure;
     declare j fixed;
     call rand;
     x=floor(rnd*10)+1;
     call rand;  do j=1 to floor(rnd*20); call rand; end;
     y=floor(rnd*10)+1;
     call rand;  do j=1 to floor(rnd*20); call rand; end;
     z=floor(rnd*10)+1;
end thcord;

where : procedure;
      put skip list ('Currently at:',position(1),position(2),
          position(3));
end where;

dragon:procedure;
   newdrag=1;
   old=position;
   put skip list ('                  ');
   put skip list ('There is a dragon in this room!');
   call rand;
   anger=floor(rnd*10)+1;
   call rand;
   hp=floor(rnd*5)+18;
   if anger<4 then put skip list ('You are lucky...it is asleep.');
   if anger>7 then do;
       put skip list ('It heard you coming and has sneak attacked!');
       call rand;
       if floor(rnd*11)>6 then do;
           put skip list('You are badly bitten!');
           call rand; dead=dead-floor(rnd*10)-21; call chesp;
          end;
          else if floor(rnd*11)<3 then
             put skip list ('It jumps at you but misses!');
              else do;
                   call rand;
                   gold=floor(gold*rnd);
                   put skip list ('It swings at you with a claw, but');
               put skip list('misses. Unfortunately, your gold purse');
               put skip list('is cut open and you lose some gold!');
              end;
   end;
   if anger<8 then if anger>3 then do;
      put skip list ('You have both surprised one another,');
      call rand;
      if floor(rnd*11)>5 then
         put skip list ('however, you gain the initiative!');
      else do;
            put skip list ('but it acts first!');
            call rand;
            if floor(rnd*11)>3 then call dturn;
            else do;
            call rand;
            put skip list ('The dragon roars ferociously at you,');
            put skip list('shattering some of your possessions!');
            call rand;
            if pos_sub>5 then pos_sub=floor(pos_sub*rnd)+1;
               else if pos_sub>1 then pos_sub=pos_sub-1;
            end;
      end;
   end;
end dragon;

search:  procedure;
     declare temp picture '999';
     flg1=1;
     temp=m(position(1),position(2),position(3));
     if temp=1 then
          put skip list ('You can''t, while a dragon''s present.');
     if temp=2 then put skip list ('There is gold here!');
     if temp=3 then put skip list ('There is a potion here!');
     if temp=4 then put skip list ('There is a strange wand here!');
     if temp=5 then put skip list ('There is a portal here!');
     if temp=6 then put skip list ('There is a portal here!');
     if temp=7 then put skip list ('There is a sword here!');
     if temp=0 then put skip list ('There is nothing here.');
end search;

untrap:procedure;
   if ckfl=0 then put skip list ('You have to find a trap first.');
   else do;
   if t(position(1),position(2),position(3))=0 then
      put skip list ('There is no trap here.');
   else do;
           call rand;
           if floor(rnd*10)+1>2 then do;
              put skip list ('You are successful.');
              added=added+3;
           end;
           else do;
              put skip list ('You accidentally spring the ');
              put skip list ('trap on yourself and...');
              call rand;
              if floor(rnd*10)<3 then
                  put skip list ('are injured by falling rocks!');
              else if floor(rnd*10)<6 then
                  put skip list ('are injured by flying rocks!');
              else if floor(rnd*10)<8 then
                  put skip list ('are bitten by a snake in a pit!');
              else do;
                  put skip list ('are crushed as the floor caves');
                  put skip list ('in, but you do not fall through!');
              end;
              dead=dead-floor(rnd*4)-1;
              added=added+1;
           end;
           trap=0;
           t(position(1),position(2),position(3))=0;
   end;
   end;
end untrap;

abbrev2:procedure;
   declare c character (10) varying;
   c=item;
   if index('north',c)=1 then c='north';
   else if index('south',c)=1 then c='south';
   else if index('east',c)=1 then c='east';
   else if index('west',c)=1 then c='west';
   else if index('up',c)=1 then c='up';
   else if index('down',c)=1 then c='down';
   item=c;
end abbrev2;

move : procedure;
     declare (flag,flag1) fixed;
     flag=0; flag1=0;
 l1: put skip list ('What direction?');
     get skip list (item);
     call abbrev2;
     if item='up' then if position(3)=1 then flag=1;
        else do; position(3)=position(3)-1; flag1=1; end;
     if item='down' then if  position(3)=10 then flag=1;
        else do; position(3)=position(3)+1; flag1=1; end;
     if item='north' then if  position(1)=1 then flag=1;
        else do; position(1)=position(1)-1; flag1=1; end;
     if item='south' then if  position(1)=10 then flag=1;
        else do; position(1)=position(1)+1; flag1=1; end;
     if item='west' then if  position(2)=1 then flag=1;
        else do; position(2)=position(2)-1; flag1=1; end;
     if item='east' then if  position(2)=10 then flag=1;
        else do; position(2)=position(2)+1; flag1=1; end;
     if flag=1 then put skip list ('You can''t go',item,'! ');
        else if flag1=0 then goto l1;
     if flag=1 then ;
         else do;
            ckfl=0; flg1=0;
            put skip list ('Ok.');
            if t(position(1),position(2),position(3))=1 then trap=1;
            else trap=0;
         end;
     if m(position(1),position(2),position(3))=1 then
         if flag1=1 then call dragon;
end move;

take : procedure;
     declare i picture '9';
     declare flag fixed;
     flag=0;
     if flg1=0 then
        put skip list ('Take what? You haven''t searched yet.');
     else do;
        if m(position(1),position(2),position(3))=4 then do;
             flag=1;
             m(position(1),position(2),position(3))=0;
             possess(pos_sub)='wand';
             pos_sub=pos_sub+1;
        end;
        if m(position(1),position(2),position(3))=7 then do;
           flag=1;
           m(position(1),position(2),position(3))=0;
           possess(pos_sub)='sword';
           pos_sub=pos_sub+1;
        end;
        if m(position(1),position(2),position(3))=1 then do;
                  put skip list ('Be serious!!');
                  flag=3;
        end;
        if m(position(1),position(2),position(3))=3  then do;
          flag=1;
          m(position(1),position(2),position(3))=0;
          possess(pos_sub)='potion';
          pos_sub=pos_sub+1;
        end;
        if m(position(1),position(2),position(3))=2  then do;
          call rand; i=floor(rnd*5)+1;
          gold=gold+i;
          put skip list (i,'gold pieces');
          put skip list ('        ');
          flag=1;
          m(position(1),position(2),position(3))=0;
         end;
     if m(position(1),position(2),position(3))=5 then do;
         put skip list ('You can''t possibly take a portal!');
         flag=3;
     end;
     if m(position(1),position(2),position(3))=6 then do;
          put skip list ('You can''t possibly take a portal!');
          flag=3;
     end;
     if flag=0 then put skip list ('There is nothing here.');
          else if flag=1 then put skip list ('Ok.');
   end;
end take;

help : procedure;
     put skip list ('Valid commands are:');
     put skip list ('   attack     - attack dragon by hands or with sword');
     put skip list ('   checktraps - check cavern for a trap');
     put skip list ('   esp        - view current level');
     put skip list ('   forget     - erase an item from memory');
     put skip list ('   help       - list commands and their functions');
     put skip list ('   hide       - hide gold in cavern');
     put skip list ('   items      - take inventory of your possessions');
     put skip list ('   memory     - list contents of your memory');
     put skip list ('   move       - move to adjacent cavern');
     put skip list ('   portal     - enter portal');
     put skip list ('   potion     - drink a potion');
     put skip list ('   quit       - quit the game');
     put skip list ('   retrieve   - retrieve gold that you have hidden');
     put skip list ('   remember   - enter something into your memory');
     put skip list ('   search     - search cavern for items');
     put skip list ('   status     - check energy level, esp, and score ');
     put skip list ('   take       - take an item from cavern');
     put skip list ('   untrap     - unset a trap in cavern');
     put skip list ('   wand       - use a wand against dragon');
     put skip list ('   where      - list x,y,z coordin. of position in dungeon');
end help;

items: procedure;
      do x=1 to pos_sub-1;
         put skip list (x,possess(x));
      end;
      put skip list ('                         ');
      put skip list (gold,'gold pieces.');
end items;

lev : procedure;
   declare flag fixed;
   if okesp<=0 then do;
      put skip list (' ');
      put list ('1      2      3      4      5      6');
      put list ('7      8      9      10');
      do x=1 to 10;
      put skip list (x);
      do y=1 to 10;
      flag=0;
      if x=position(1) then if y=position(2) then do;
           put list ('@'); flag=1; end;
      if flag=0
      then if m(x,y,position(3))>0 then put list('*');
          else put list('.');
      end;
      end;
   end;
   else do; put skip list ('The last injury you suffered');
       put skip list ('has messed up your esp powers.'); end;
end lev;

portal:procedure;
  if flg1=0 then put skip list ('You haven''t searched for a portal here yet.');
  else
   if m(position(1),position(2),position(3))=6
   then do;
      put skip list ('You enter the portal and there is a');
      put skip list ('blinding flash of light.');
      put skip list ('The next thing you know...');
      put skip list ('                ');
      put skip list ('You''re outside in the sunlight in the');
      put skip list ('middle of a grassy meadow!');
      put skip list ('         ');
      put skip list ('You made it!!!');
      call score;
      added=added+50;
      put skip list ('          ');
      put skip list ('You vanquished',drag,'dragons, and');
      put skip list ('with your gold and other possessions...');

      command='quit';
   end;
   else if m(position(1),position(2),position(3))=5
   then do;
      put skip list ('You enter the portal and there is a');
      put skip list ('blinding flash of light!');
      put skip list ('The next thing you know...');
      put skip list ('           ');
      put skip list ('Nothing else happens!');
      put skip list ('              ');
      put skip list ('This is a fake portal!');
      m(position(1),position(2),position(3))=0;
   end;
   else put skip list ('There is no portal here.');
end portal;

wand:procedure;
   declare (w,flag,temp) fixed (4);
   flag=0;
   if m(position(1),position(2),position(3))=1
   then do;
      put skip list ('Which number wand will you use?');
      call getnum; w=number;
      if w>0 then if w<pos_sub then if possess(w)='wand'
      then do;
         flag=1;
         put skip list ('You get out wand #',w,'and wave');
         put skip list ('it high over your head.');
         put skip list ('        ');
         call rand;
         temp=floor(rnd*10)+1;
         if temp>8 then do;
           put skip list ('The dragon suddenly turns red with anger!');
           anger=anger+5;
         end;
         else if temp>6 then do;
           put skip list ('The dragon suddenly disappears!');
           call replace;
           anger=0;
           drag=drag+1;
         end;
         else if temp>4 then do;
           put skip list ('The dragon explodes, splattering its');
           put skip list ('guts all over the room...and you!');
           do i=1 to floor(rnd*20)+1; call rand; end;
           if floor(rnd*10)+1>5 then do;
               call rand;
               dead=dead-floor(rnd*5)-1;
               put skip list ('Its blood seems to burn your skin');
               put skip list ('a bit, but then it stops.');
           end;
           anger=0;
           call replace;
           drag=drag+1;
         end;
         else if temp>2 then do;
           put skip list ('The dragon suddenly falls to ');
           put skip list ('the ground! Then it disappears!');
           anger=0;
           call replace;
           drag=drag+1;
         end;
         else do;
           put skip list ('The dragon begins to smile!');
           call rand;
           temp=floor(rnd*5)+1;
           put skip list ('It befriends you and gives you');
           put skip list (temp,'gold pieces!');
           put skip list ('Then it flies away.');
           anger=0;
           call replace;
           drag=drag+1;
           gold=gold+temp;
         end;
         put skip list ('The wand disappears from your hand.');
         pos_sub=pos_sub-1;
         possess(w)=possess(pos_sub);
      end;
      if flag=0 then put skip list ('You have no wand #',w,'.');
   end;
   else put skip list ('Wands only work in the presence of dragons.');
end wand;

potion:procedure;
   declare (p,flag,temp) fixed (4);
   flag=0;
   put skip list ('Which number potion will you drink?');
   call getnum; p=number;
   if p>0 then if p<pos_sub then if possess(p)='potion'
   then do;
      flag=1;
      put skip list ('You drink the potion and...');
      call rand;
      temp=floor(rnd*10)+1;
      if temp>5 then do;
         put skip list ('you feel greatly refreshed!');
         if dead<lim  then do;
            call rand;
            dead=dead+floor(rnd*20)+41;
         end;
      end;
      else if temp>3 then do;
         if okesp>0 then do;
            put skip list ('your esp is restored!');
            okesp=0;
         end;
         else put skip list ('Nothing happens!');
      end;
      else if temp>1 then do;
         put skip list ('you suddenly disappear,');
         call thcord;
         anger=0;  ckfl=0;  flg1=0;
         position(1)=x; position(2)=y; position(3)=z;
         put skip list ('and reappear at:',x,y,z);
         if t(x,y,z)=1 then trap=1; else trap=0;
         if m(x,y,z)=1 then call dragon;
      end;
      else do;
         put skip list ('you get great pains in your stomach!');
         put skip list ('It was poison!');
         call rand;
         dead=dead-floor(rnd*15)-6;
      end;
      pos_sub=pos_sub-1;
      possess(p)=possess(pos_sub);
   end;
   if flag=0 then put skip list ('You have no potion #',p,'.');
end potion;

dturn:procedure;
  declare flag fixed;
  flag=0;
  put skip list ('            ');
  if hp<=0 then do;
   put skip list ('The dragon falls dead!');
   put skip list ('And then it disappears.');
   anger=0;
   drag=drag+1;
   call replace;
  end;
  else do;
   if anger<10 then anger=anger+1;
   if m(position(1),position(2),position(3))=1
   then do;
      if anger<5 then do;
         call rand;
         if floor(rnd*11)<5 then do;
            put skip list ('The dragon jumps to its feet.');
            anger=anger+4;
         end;
         else if floor(rnd*11)<8 then do;
            put skip list ('The dragon must have been fast');
            put skip list ('asleep because it is only slowly');
            put skip list ('rising.');
            anger=anger+1;
         end;
         else do;
            put skip list ('The dragon wakes up and flies away!');
            anger=0;
            call replace;
            drag=drag+1;
         end;
      end;
      else if anger>7 then do;
         call rand;
         if floor(rnd*11)>2 then do;
             call chesp;
             put skip list ('The dragon attacks you!');
             call rand;
             if floor(rnd*11)>9 then do;
                put skip list ('The dragon knocks you in the head');
                put skip list ('and you lose some of your memory!');
                call chesp;
                if mempoint>1 then do;
                   call rand;
                   i=floor((mempoint-1)*rnd)+1;
                t1:call thcord;
                   if m(x,y,z)=0 then do;
                      m(x,y,z)=m(memw(i,1),memw(i,2),memw(i,3));
                      m(memw(i,1),memw(i,2),memw(i,3))=0;
                   end;
                   else goto t1;
                   do x=1 to hidpoint;
                      if hidden(x,2)=memw(i,1) then
                      if hidden(x,3)=memw(i,2) then
                      if hidden(x,4)=memw(i,3) then do;
                         flag=1;
                         hidpoint=hidpoint-1;
                         hidden(x,1)=hidden(hidpoint,1);
                         hidden(x,2)=hidden(hidpoint,2);
                         hidden(x,3)=hidden(hidpoint,3);
                         hidden(x,4)=hidden(hidpoint,4);
                      end;
                   end;
                   if flag=0 then
                      if hidpoint>1 then hidpoint=hidpoint-1;
                   mempoint=mempoint-1;
                   mem(i)=mem(mempoint);
                   memw(i,1)=memw(mempoint,1);
                   memw(i,2)=memw(mempoint,2);
                   memw(i,3)=memw(mempoint,3);
                end;
             end;
             else if floor(rnd*11)>6 then do;
                put skip list ('It breathes fire and burns you!');
                call rand;
                dead=dead-floor(rnd*10)-21;
             end;
             else if floor(rnd*11)>3 then do;
                put skip list ('It crushes you against the wall!');
                call rand;
                dead=dead-floor(rnd*10)-21;
             end;
             else if floor(rnd*11)>1 then do;
                put skip list ('It misses you, but spills your gold!');
                call rand;
                gold=floor(gold*rnd);
             end;
             else do;
                put skip list ('It claws you, breaking some of your');
                put skip list ('possessions.');
                call rand;
                if pos_sub>5 then pos_sub=floor(pos_sub*rnd)+1;
                    else if pos_sub>1 then pos_sub=pos_sub-1;
                call rand;
                dead=dead-floor(rnd*10)-1;
             end;
          end;
          else do;
             put skip list ('The dragon does not attack you but');
             put skip list ('flies in circles above your head!');
             do i=1 to floor(rnd*20)+1; call rand; end;
             if floor(rnd*10)+1>6 then do;
                call chesp;
                put skip list ('You get dizzy watching it and you');
                put skip list ('fall...hurting yourself!');
                call rand;
                dead=dead-floor(rnd*5)-1;
             end;
          end;
      end;
      else do;
         call rand;
         if floor(rnd*11)<2 then do;
            put skip list ('The dragon turns and flies away!');
            anger=0;
            call replace;
            drag=drag+1;
         end;
         else if floor(rnd*11)<5 then
            put skip list ('The dragon now pauses, studying you.');
         else if floor(rnd*11)<9 then do;
            call chesp;
            put skip list ('The dragon rips your flesh with its');
            put skip list ('huge claws!');
            call rand;
            dead=dead-floor(rnd*10)-21;
         end;
         else do;
            put skip list ('The dragon pounces on you, destroying');
            put skip list ('every one of your possessions...and');
            put skip list ('nearly destroying you!');
            call rand;
            dead=dead-floor(rnd*10)-21;
            pos_sub=1;
            call chesp;
         end;
      end;
   end;
   else do;
      call rand;
      anger=anger-4;
      if anger<4 then do;
         put skip list ('You no longer see nor hear the dragon.');
         anger=0;
      end;
      else do;
         put skip list ('The dragon has followed you!');
         m(old(1),old(2),old(3))=0;
         old=position;
         m(position(1),position(2),position(3))=1;
         call rand;
         if floor(rnd*11)<3 then do;
            anger=anger+1;
            call dturn;
         end;
         else if floor(rnd*11)<6 then do;
            put skip list ('It bites you from behind!');
            call rand; call chesp;
            dead=dead-floor(rnd*10)-11;
         end;
         else if floor(rnd*11)<8 then
            put skip list ('It breathes fire at you, but misses!');
         else
            put skip list ('But you are ready for it!');
      end;
   end;
  end;
end dturn;

attack:procedure;
   declare (temp,flag) fixed (4);
   declare p character (10) varying;
   if m(position(1),position(2),position(3))=1
   then do;
  l1:put skip list ('With what?');
     get skip list (p);
     if index('hands',p)=1 then p='hands';
     else if index('sword',p)=1 then p='sword';
     if p='hands'
     then do;
        put skip list ('You attempt to strike the dragon...');
        call rand;
        temp=floor(rnd*10)+1;
        if temp<3 then
           put skip list ('but it easily dodges your blow!');
        else if temp<6 then do;
           put skip list ('and you land a hard right to its snout!');
           call rand; hp=hp-floor(rnd*4)-4;
        end;
        else if temp<8 then do;
           put skip list ('and you really lay one on it!');
           call rand; hp=hp-floor(rnd*4)-6;
        end;
        else do;
            put skip list ('you kick the beast in the stomach!');
            call rand; hp=hp-floor(rnd*4)-4;
        end;
     end;
     else if p='sword'
     then do;
       flag=0;
       do i=1 to pos_sub-1;
          if possess(i)='sword' then do; flag=1; u=i; end;
       end;
       if flag=0 then do;
          put skip list ('You have no sword.');
          put skip list ('           ');
          goto l1;
       end;
       else do;
          put skip list ('You whip out your mighty sword and');
          put skip list ('slash at the dragon...');
          call rand;
          temp=floor(rnd*10)+1;
          if temp<3 then do;
             put skip list ('deeply wounding it!');
             call rand;  hp=hp-floor(rnd*5)-8;
          end;
          else if temp<5 then
             put skip list ('but you miss it!');
          else if temp<8 then do;
             put skip list ('and draw much blood!');
             call rand;  hp=hp-floor(rnd*5)-8;
          end;
          else if temp=8 then do;
             put skip list ('your sword breaks!');
             pos_sub=pos_sub-1;
             possess(u)=possess(pos_sub);
          end;
          else do;
             put skip list ('hitting it, but not doing');
             put skip list ('too much harm!');
             call rand; hp=hp-floor(rnd*4)-6;
          end;
       end;
     end;
     else do;
        put skip list ('Hands or sword only.');
        put skip list ('              ');
        goto l1;
     end;
   end;
   else put skip list ('There is no dragon here.');
end attack;

remember:procedure;
     declare p character (10) varying;
     if mempoint=11 then
        put skip list ('Your memory is full.');
     else do;
        put skip list ('What do you wish to remember?');
        get skip list (p);
        mem(mempoint)=p;
        put skip list ('At what coordinates?');
        put skip list ('   first  :');
        call getnum; memw(mempoint,1)=number;
        put  list     ('   second :');
        call getnum; memw(mempoint,2)=number;
        put  list     ('   third  :');
        call getnum; memw(mempoint,3)=number;
        mempoint=mempoint+1;
        put skip list ('Ok.');
      end;
end remember;


memory:procedure;
   do x=1 to mempoint-1;
      put skip list (x,':',mem(x),'at',memw(x,1),memw(x,2),memw(x,3));
   end;
end memory;

forget:procedure;
   if mempoint=1 then
      put skip list ('There''s nothing in your memory.');
   else do;
      put skip list ('Which number memory item do you wish to forget?');
      call getnum; x=number;
      if x>0 then if x<mempoint then do;
         mempoint=mempoint-1;
         mem(x)=mem(mempoint);
         memw(x,1)=memw(mempoint,1);
         memw(x,2)=memw(mempoint,2);
         memw(x,3)=memw(mempoint,3);
         put skip list ('Ok.');
      end;
      else put skip list ('You have no memory item #',x,'.');
   end;
end forget;

hide:procedure;
   declare p fixed (4);
   if hidpoint=101 then
      put skip list ('You can''t hide in more than 100 locations.');
   else do;
         put skip list ('How many gold pieces do you wish to hide?');
         call getnum;
         p=number;
         if p>gold then
             put skip list ('You only have',gold,'gold pieces.');
         else if p<0 then
             put skip list ('Don''t be cute.');
         else do;
            hidden(hidpoint,1)=p;
            hidden(hidpoint,2)=position(1);
            hidden(hidpoint,3)=position(2);
            hidden(hidpoint,4)=position(3);
            gold=gold-p;
            hidpoint=hidpoint+1;
            put skip list ('Ok.');
         end;
   end;
end hide;

retrieve:procedure;
    declare (p,flag) fixed (4);
    flag=0;
    p=1;
 l1:if p<hidpoint then do;
    if position(1)=hidden(p,2) then if position(2)=hidden(p,3)
       then if position(3)=hidden(p,4)
       then do;
          flag=1;
          gold=gold+hidden(p,1);
          put skip list ('You''ve retrieved',hidden(p,1),'gold pieces.');
          hidpoint=hidpoint-1;
          hidden(p,1)=hidden(hidpoint,1);
          hidden(p,2)=hidden(hidpoint,2);
          hidden(p,3)=hidden(hidpoint,3);
          hidden(p,4)=hidden(hidpoint,4);
       end;
    p=p+1;
    goto l1;
    end;
    if flag=0 then
       put skip list ('There is nothing hidden here.');
end retrieve;

falltrap:procedure;
   call rand;
   if floor(rnd*10)+trap+1<7 then
      trap=trap+1;
   else do;
      trapflag=1;
      trap=0;
      t(position(1),position(2),position(3))=0;
      put skip list ('Unfortunately, as you are ');
      put skip list ('beginning to',command,'...');
      call rand;  dead=dead-floor(rnd*7)-1;
   g1:call rand; i=floor(rnd*11);
      if i>8 then do;
         put skip list ('a bunch of rocks fall from');
         put skip list ('the ceiling and injure you!');
      end;
      else if i>6 then do;
         put skip list ('a spear comes flying out of');
         put skip list ('nowhere and pierces your side!');
      end;
      else if i>4 then do;
         put skip list ('you trip over a small cord that was');
         put skip list ('stretched across the floor, and fall');
         put skip list ('into a shallow depression full of');
         put skip list ('poisonous snakes and are bitten!');
      end;
      else if i>2 then do;
         put skip list ('you step on a rock which moves, and');
         put skip list ('all of a sudden you are wounded by');
         put skip list ('a bunch of flying darts!');
      end;
      else if position(3)<10 then do;
         call rand;
         v=floor(rnd*(10-position(3)))+1;
         put skip list ('the floor caves in and you');
         put skip list ('fall down',v,'levels!');
         position(3)=position(3)+v; ckfl=0;  flg1=0;
         if t(position(1),position(2),position(3))=1 then trap=1;
         if m(position(1),position(2),position(3))=1 then call dragon;
      end;
      else goto g1;
      call rand;
      if floor(rnd*10)+1>5 then do;
         put skip list ('You are badly hurt!');
         call chesp;
         dead=dead-3;
      end;
   end;
end falltrap;

checktrap:procedure;
   if anger>0 then
      put skip list ('You can''t while a dragon''s present!');
   else do;
   if t(position(1),position(2),position(3))=0
   then put skip list ('There is no trap here.');
   else
      put skip list ('Beware! There''s a trap!');
   ckfl=1;
   end;
end checktrap;
