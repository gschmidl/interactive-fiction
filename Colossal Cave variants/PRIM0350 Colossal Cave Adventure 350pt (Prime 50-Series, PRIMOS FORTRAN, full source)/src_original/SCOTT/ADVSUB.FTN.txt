      SUBROUTINE DATIME(D,T)
C
C  RETURN THE CURRENT DATE AND TIME
C
C  D IS SET TO THE NUMBER OF DAYS SINCE 01/01/77 (A SATURDAY)
C  T IS SET TO THE NUMBER OF MINUTES SINCE MIDNIGHT
C
C
      IMPLICIT INTEGER (A-Z)
      REAL*4 DOFY$A, TIME$A, DINT, TEMP, YD
C
      YD = DOFY$A(TEMP) - 77.001E0
      N = INT(YD)
      NM = (YD - AINT(YD)) * 1000.E0
      D = 1461 * (N/4) + 365 * MOD(N,4) + NM + 1
      T = TIME$A(TEMP) * 60.
C
      RETURN
      END

      INTEGER FUNCTION SHIFT(VAL,DIST)
      IMPLICIT INTEGER(A-Z)
C
C   RETURN VAL LEFT SHIFTED (LOGICALLY) DIST BITS (RIGHT SHIFT IF DIST<0)
C
      IF( DIST .GE. 0 ) SHIFT = LS( VAL, DIST )
      IF( DIST .LT. 0 ) SHIFT = RS( VAL,-DIST )
C
      RETURN
      END

      SUBROUTINE HOURSX(H,DAYTYP)
C
C  USED BY HOURS (ABOVE) TO PRINT HOURS FOR EITHER WEEKDAYS OR WEEKENDS.
C
      IMPLICIT INTEGER(A-Z)
      LOGICAL FIRST,BLKLIN
      COMMON /IOSCOM/ TTYI,TTYO,BLKLIN,DBFI
C
      DIMENSION TYPE(10,3)
      DATA TYPE
     1     /1HM,1HO,1HN,1H ,1H-,1H ,1HF,1HR,1HI,1H:,
     2      1HS,1HA,1HT,1H ,1H&,1H ,1HS,1HU,1HN,1H:,
     3      1HH,1HO,1HL,1HI,1HD,1HA,1HY,1HS,1H:,1H /
C
C
      FIRST=.TRUE.
      FROM=-1
      IF(H.NE.0)GOTO 10
C
      WRITE(TTYO,2)(TYPE(J,DAYTYP),J=1,10)
    2 FORMAT(10X,10A1,'   open all day')
      RETURN
C
   10 FROM=FROM+1
      IF(AND(H,SHIFT(1,FROM)).NE.0)GOTO 10
      IF(FROM.GE.24)GOTO 20
      TILL=FROM
   14 TILL=TILL+1
      IF(AND(H,SHIFT(1,TILL)).EQ.0.AND.TILL.NE.24)GOTO 14
C
      IF(FIRST)WRITE(TTYO,16)(TYPE(J,DAYTYP),J=1,10),FROM,TILL
      IF(.NOT.FIRST)WRITE(TTYO,18)FROM,TILL
   16 FORMAT(10X,10A1,I4,':00 to',I3,':00')
   18 FORMAT(20X,I4,':00 to',I3,':00')
      FIRST=.FALSE.
      FROM=TILL
      GOTO 10
C
   20 IF(FIRST)WRITE(TTYO,22)(TYPE(J,DAYTYP),J=1,10)
   22 FORMAT(10X,10A1,'   closed all day')
      RETURN
      END

      SUBROUTINE LDCOMN(QNAME,NAME,CMADRS,CMSZES)
$INSERT SYSCOM>KEYS.F
C
C   ADVENTURE SUBROUTINE TO LOAD ALL COMMON BLOCKS FROM EITHER
C     THE SYSTEM FILE (QNAME=.TRUE.) OR A PREVIOUSLY SAVED FILE
C     (QNAME=.FALSE.)
C
C   IF THE SYSTEM FILE IS NOT FOUND, DO  NOTHING
C
C   IF THE NAMED FILE IS NOT FOUND, DIE
C
      IMPLICIT INTEGER (A-Z)
      LOGICAL BLKLIN
      COMMON /IOSCOM/ TTYI,TTYO,BLKLIN,DBFI
      LOGICAL QNAME
      DIMENSION CMADRS(4,11), CMSZES(11), NAME(10)
      INTEGER *2 SYSNAM(3), K6, K1, X, CODE, I, K32,  K10
      PARAMETER K6=6, K1=1, K32=32,  K10=10
      DATA SYSNAM/'ADVCOM'/
C
      CALL SRCH$$(K$CLOS,0,0,K1,X,CODE)
      IF( .NOT. QNAME ) GO TO 10
      CALL SRCH$$(K$READ,SYSNAM,K6,K1,X,CODE)
      IF( CODE .NE. 0 ) GO TO 35
      GO TO 20
C
   10 CALL PACK(NAME,1,10,NAME,1)
      IF( NAME(1) .EQ. '    ' ) GO TO 40
      CALL SRCH$$(K$READ,NAME,K10,K1,X,CODE)
      CALL ERRPR$(K$NRTN,CODE,NAME,K10,'LDCOMN',K6)
C
   20 DO  30  I = 1, 11
   30 CALL PRWF$$(K$READ,K1,CMADRS(1,I),INTS(CMSZES(I)),0000000,X,CODE)
C
      CALL SRCH$$(K$CLOS,0,0,K1,X,CODE)
35    CONTINUE
      RETURN
C
   40 CALL TNOUA(' What file should I search for? ',K32)
      READ(TTYI,50) NAME
   50 FORMAT(10 A1)
      GO TO 10
      END

      SUBROUTINE SVCOMN(QNAME,NAME,CMADRS,CMSZES)
$INSERT SYSCOM>KEYS.F
C
C   ADVENTURE SUBROUTINE TO SAVE COMMON FOR FUTURE USE
C
C   OPEN FILE ADVCOM (NAME) DEPENDING ON QNAME AND WRITE OUT ALL
C      PROGRAM VARIABLES FOR RESTART LATER
C
      IMPLICIT INTEGER (A-Z)
      LOGICAL BLKLIN
      COMMON /IOSCOM/ TTYI,TTYO,BLKLIN,DBFI
      LOGICAL QNAME
      DIMENSION CMADRS(4,11), CMSZES(11), NAME(10)
      INTEGER *2 SYSNAM(3), K1, K6, X, CODE, I, K10, K31, K32
      PARAMETER K1=1, K6=6, K10=10, K31=31, K32=32
      DATA SYSNAM/'ADVCOM'/
C
      CALL SRCH$$(K$CLOS,0,0,K1,X,CODE)
      IF( .NOT. QNAME ) GO TO 10
      CALL SRCH$$(K$WRIT,SYSNAM,K6,K1,X,CODE)
      IF( CODE .NE. 0 ) CALL ERRPR$(K$SRTN,CODE,0,0,0,0)
      GO TO 20
C
   10 CALL PACK(NAME,1,10,NAME,1)
      IF( NAME(1) .EQ. '    ' ) GO TO 40
      CALL SRCH$$(K$WRIT,NAME,K10,K1,X,CODE)
      IF( CODE .NE. 0 ) CALL ERRPR$(K$SRTN,CODE,0,0,0,0)
C
   20 CALL PRWF$$(K$TRNC,K1,0,0,000000,X,CODE)
C
      DO  30  I = 1, 11
   30 CALL PRWF$$(K$WRIT,K1,CMADRS(1,I),INTS(CMSZES(I)),000000,X,CODE)
C
      CALL SRCH$$(K$CLOS,0,0,K1,X,CODE)
      RETURN
C
   40 CALL TNOUA(' What name should I file under? ',K32)
      READ(TTYI,50) NAME
   50 FORMAT(10 A1)
      GO TO 10
      END

      SUBROUTINE PACK   (IFROM,ISTRT,ISTOP,JDIM,JSTRT)
      INTEGER*2 IFROM(1), JDIM(1), JCON, ICON, I
      INTEGER *4 ISTRT, ISTOP, JSTRT
C
      JCON = JSTRT + ( ISTOP - ISTRT )/2
      ICON = ISTRT
C
      DO  10  I = JSTRT, JCON
      JDIM(I) = LT( IFROM(ICON),8 ) + RS( IFROM(ICON+2),8 )
   10 ICON = ICON + 4
C
      IF( MOD(ISTOP-ISTRT+1,2) .EQ. 1 )
     1      JDIM(JCON) = LT( JDIM(JCON),8 ) + :240
      RETURN
      END

      INTEGER FUNCTION RAN(RANGE)
C
C  RETURN RANDOM UNIFORMLY DISTRIBUTED VALUE IN CLOSED INTERVAL [0,RANGE-1].
C
C  COURTESY MICHAEL GRUBIN@PRIME COMPUTER INC., FRAMINGHAM MASS.
C
C
      IMPLICIT INTEGER (A-Z)
      REAL RAND$A
      COMMON /RANCOM/ R
C
      IF( R .EQ. 0 ) CALL RNDI$A(R)
      RAN = (RANGE-1)*RAND$A(R)
      RETURN

      SUBROUTINE ULCASE (KEY,BUF,NWDS)
C
C     Pr1me Computer, Inc.
C     Copyright 1980, Pr1me Computer, Inc., Framingham, MA
C
C     R.W. Bowers
C     Pr1me Computer, Inc.
C     1970 Chain Bridge Rd.
C     McLean, Va   22101
C
C     (703) 821-8618
C
      INTEGER KEY,BUF(1),NWDS,II,JJ
      INTEGER CHRTBL(2,26)
C
      LOGICAL UPCASE
C
      DATA CHRTBL /'A','a'
     +            ,'B','b'
     +            ,'C','c'
     +            ,'D','d'
     +            ,'E','e'
     +            ,'F','f'
     +            ,'G','g'
     +            ,'H','h'
     +            ,'I','i'
     +            ,'J','j'
     +            ,'K','k'
     +            ,'L','l'
     +            ,'M','m'
     +            ,'N','n'
     +            ,'O','o'
     +            ,'P','p'
     +            ,'Q','q'
     +            ,'R','r'
     +            ,'S','s'
     +            ,'T','t'
     +            ,'U','u'
     +            ,'V','v'
     +            ,'W','w'
     +            ,'X','x'
     +            ,'Y','y'
     +            ,'Z','z'/
C
      II=1
      UPCASE=.FALSE.
C
10    CONTINUE
      IF (BUF(II).EQ.' '
     +.AND.BUF(II+1).EQ.'I'
     +.AND.BUF(II+2).EQ.' ') II=II+3             /* ' I '
      IF (BUF(II).EQ.'.'
     +.AND.BUF(II+1).EQ.' '
     +.AND.BUF(II+2).EQ.' ') II=II+4
      IF (BUF(II).EQ.'!'
     +.AND.BUF(II+1).EQ.' '
     +.AND.BUF(II+2).EQ.' ') II=II+4

      IF (BUF(II).EQ.CHRTBL(1,JJ)) BUF(II)=CHRTBL(2,JJ)
15    CONTINUE
C
      IF (II.GE.NWDS) GO TO 20
      II=II+1
      GO TO 10
C
20    CONTINUE
      RETURN
