       SUBROUTINE HOURS(class)

C  Announce the current hours when the cave is open for adventuring.
C  This info is stored in WKDAY, WKEND, and HOLID, where bit SHIFT(1,N)
C  is on if the hour from N:00 to N:59 is "prime time" (cave closed).
C  WKDAY is for weekdays, WKEND for weekends, and HOLID for holidays.
C  The next holiday is from HBEGIN to HEND, and its name is HNAME.

C  Two types (classes) of players are occomodated.  Paying customers
C  (class 1) normally have freer access to the cave than in-house
C  (class 2) users, since they presumably pay for the privilege of
C  adventuring.

C  This routine is passed one argument, a "class" number indicating
C  what type of adventurer we've got here.  Users are not told which
C  class they are in, or even that there is another class.

       IMPLICIT INTEGER(A-Z)
      include comwiz                    @ define WIZCOM common block
      character*5 day                                                   DK

       PRINT 1
1      FORMAT ()
       CALL HOURSX(WKDAY(class),'Mon -',' Fri:')                        DK
       CALL HOURSX(WKEND(class),'Sat -',' Sun:')                        DK
       CALL HOURSX(HOLID(class),'Holid','ays: ')                        DK
       CALL DATIME(D,T)
       IF(HEND.LT.D.OR.HEND.LT.HBEGIN)RETURN
       IF(HBEGIN.GT.D)GOTO 10
       PRINT 5, HNAME
5      FORMAT (/' Today is a holiday, namely ',4A5)
       goto 20                                                          DK

10     D=HBEGIN-D
       day='days,'                                                      DK
       IF (D.EQ.1) day='day, '                                          DK
       PRINT 15, D,DAY,HNAME                                            DK
15     FORMAT (/' The next holiday will be in',I3,' ',A5,' namely ',4A5)
20     RETURN


       SUBROUTINE HOURSX(H,DAY1,DAY2)

C  Subroutine used by HOURS (above) to print cave hours for weekdays,
C  weekends, or holidays.

       IMPLICIT INTEGER(A-Z)
       LOGICAL FIRST
       character*5 day1,day2                                            DK

       FIRST=.TRUE.
       FROM=-1
       IF (H.NE.0) GOTO 110                                             DK
       PRINT 102, DAY1,DAY2                                             DK
102    FORMAT (10X,2A5,'   Open all day')                               DK
       goto 130                                                         DK

110    FROM=FROM+1                                                      DK
       IF (AND(H,SHIFT(1,FROM)).NE.0) GOTO 110                          DK
       IF (FROM.GE.24) GOTO 120                                         DK
       TILL=FROM
114    TILL=TILL+1
       IF (AND(H,SHIFT(1,TILL)).EQ.0.AND.TILL.NE.24) GOTO 114           DK
       IF (FIRST) PRINT 116, DAY1,DAY2,FROM,TILL                        DK
       IF (.NOT.FIRST) PRINT 118, FROM,TILL                             DK
116    FORMAT (10X,2A5,I4,':00 to',I3,':00')                            DK
118    FORMAT (20X,I4,':00 to',I3,':00')                                DK
       FIRST=.FALSE.
       FROM=TILL
       GOTO 110                                                         DK

120    IF (FIRST) PRINT 122, DAY1,DAY2                                  DK
122    FORMAT (10X,2A5,'  Closed all day')                              DK
130    RETURN                                                           DK

       END



       SUBROUTINE NEWHRS

C  Set up new hours for the cave.  Specified as inverse--i.e., when is it
C  closed due to prime time?  See HOURS (above) for desc of variables.

       IMPLICIT INTEGER(A-Z)
      include comwiz                    @ define WIZCOM common block
      character*8 clsmsg                                                DK

       CALL MSPEAK(21)
       do 20 clsnum=1,2                                                 DK
       clsmsg = 'paying'                                                DK
       if (clsnum.eq.2) clsmsg = 'in-house'                             DK
       print 10, clsmsg                                                 DK
10     format (/,' New hours for ',A8,' adventurers:')                  DK
       WKDAY(clsnum)=NEWHRX('Weekd','ays:')                             DK
       WKEND(clsnum)=NEWHRX('Weeke','nds:')                             DK
       HOLID(clsnum)=NEWHRX('Holid','ays:')                             DK
       CALL MSPEAK(22)
       call hours(clsnum)                                               DK
20     continue                                                         DK
       RETURN


       INTEGER FUNCTION NEWHRX(DAY1,DAY2)
       IMPLICIT INTEGER(A-Z)
       character*5 day1,day2                                            DK

C  Input prime time specs and set up a word of internal format.

       NEWHRX=0
       PRINT 110, DAY1,DAY2
110    FORMAT (' Prime time on ',2A5)
120    PRINT 130
130    FORMAT (' From:')
       READ 140, FROM
140    FORMAT ()
       IF (FROM.LT.0.OR.FROM.GE.24) goto 199                            DK
       PRINT 150
150    FORMAT (' Till:')
       READ 140, TILL
       TILL=TILL-1
       IF (TILL.LT.FROM.OR.TILL.GE.24) goto 199                         DK
       DO 160 I=FROM,TILL
160    NEWHRX=OR(NEWHRX,SHIFT(1,I))
       GOTO 120
199    return                                                           DK

       END
