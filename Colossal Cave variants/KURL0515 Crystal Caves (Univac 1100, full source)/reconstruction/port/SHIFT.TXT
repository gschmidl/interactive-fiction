C  UTILITY ROUTINES (SHIFT, RAN, DATIME, CIAO, BUG)


       INTEGER FUNCTION SHIFT(VAL,DIST)
       IMPLICIT INTEGER(A-Z)
      DATA MASK1,MASK2,MASK3/O200000000000,O377777777777,O177777777777/
      DATA MASK4/O400000000000/

C  RETURN VAL LEFT-SHIFTED (LOGICALLY) DIST BITS (RIGHT-SHIFT IF DIST<0).

       SHIFT=VAL
       IF(DIST)10,20,30
10     IDIST=-DIST
       DO 11 I=1,IDIST
       J=0
       IF(SHIFT.LT.0)J=MASK1                                             CBC
11    SHIFT=AND(SHIFT,MASK2)/2+J
20     RETURN
30     DO 31 I=1,DIST
       J=0
      IF(AND(SHIFT,MASK1).NE.0)J=MASK4
31    SHIFT=AND(SHIFT,MASK3)*2+J
       RETURN
       END
