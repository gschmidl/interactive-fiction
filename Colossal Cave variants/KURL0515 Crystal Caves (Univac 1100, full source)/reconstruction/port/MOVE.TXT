      SUBROUTINE MOVE(OBJECT,WHERE)
C
C  PLACE ANY OBJECT ANYWHERE BY PICKING IT UP AND DROPPING IT.
C  MAY ALREADY BE TOTING, IN WHICH CASE THE CARRY IS A NO-OP.
C
C  TO MOVE THE SECOND PLACE OF AN OBJECT, USE THE OBJECT'S NUMBER PLUS
C  100 AS THE OBJECT.  (MOVE MAY BE CALLED REGARDLESS OF WHERE THE
C  OBJECT WAS, WHETHER IT'S BEING CARRIED, HAS BEEN DESTROYED, ETC.)
C
      include defs                      @
      include compla                    @
C
      IF (OBJECT.GT.objsiz) GOTO 10
      FROM=PLACE(OBJECT)
      GOTO 20
C
   10 FROM=FIXED(OBJECT-objsiz)
   20 IF (FROM.GT.0 .AND. FROM.LE.300) CALL CARRY(OBJECT,FROM)
C
C     NOW DROP IT
C
      IF (OBJECT.GT.objsiz) GOTO 50
      IF (PLACE(OBJECT).EQ.-1) HOLDNG=HOLDNG-1
      PLACE(OBJECT)=WHERE
      GOTO 60
C
   50 FIXED(OBJECT-objsiz)=WHERE
   60 IF (WHERE.LE.0) RETURN
      LINK(OBJECT)=ATLOC(WHERE)
      ATLOC(WHERE)=OBJECT
      RETURN
      END


      SUBROUTINE CARRY(OBJECT,WHERE)
C
C  START TOTING AN OBJECT, REMOVING IT FROM THE LIST OF THINGS AT ITS
C  FORMER LOCATION.  INCR HOLDNG UNLESS IT WAS ALREADY BEING TOTED.
C  IF OBJECT>objsiz (MOVING "FIXED" SECOND LOC), DON'T CHANGE PLACE OR
C  HOLDNG.
C
      include defs                      @
      include compla                    @
C
      IF (OBJECT.GT.objsiz) GOTO 5
      FROM=PLACE(OBJECT)
      PLACE(OBJECT)=-1
      IF (FROM.NE.-1) HOLDNG=HOLDNG+1
      GOTO 6
C
    5 FROM=FIXED(OBJECT-objsiz)
      FIXED(OBJECT-objsiz)=-1
    6 IF (FROM.NE.WHERE) CALL BUG(34)
      IF (FROM.EQ.-1) RETURN
      TEMP=ATLOC(FROM)
      IF (OBJECT .NE. TEMP) GOTO 7
      ATLOC(FROM)=LINK(OBJECT)
      RETURN
C
    7 IF (OBJECT .EQ. LINK(TEMP)) GOTO 8
      TEMP=LINK(TEMP)
      IF (TEMP .LE. 0) CALL BUG(30)
      GOTO 7
C
    8 LINK(TEMP)=LINK(OBJECT)
      LINK(OBJECT)=0
      RETURN
      END


       SUBROUTINE DSTROY(OBJECT)

C  Permanently eliminate OBJECT by moving it to a non-existent location.

       IMPLICIT INTEGER(A-Z)

       CALL MOVE(OBJECT,0)
       RETURN
       END


      SUBROUTINE JUGGLE(OBJECT)
C
C  JUGGLE AN OBJECT BY PICKING IT UP AND PUTTING IT DOWN AGAIN, THE
C  PURPOSE BEING TO GET THE OBJECT TO THE FRONT OF THE CHAIN OF THINGS
C  AT ITS LOC.
C
      include defs                      @
      include compla                    @
C
      WHERE=PLACE(OBJECT)
      IF (OBJECT.GT.objsiz) WHERE=FIXED(OBJECT-objsiz)
      IF (WHERE.LE.0 .OR. WHERE.GT.300) CALL BUG(33)
      CALL MOVE(OBJECT,WHERE)
      RETURN
      END
