C       INTEGER FUNCTION RAN(RANGE)
C
C  SINCE THE RAN FUNCTION IN LIB40 SEEMS TO BE A REAL LOSE, WE'LL USE ONE OF
C  OUR OWN.  IT'S BEEN RUN THROUGH MANY OF THE TESTS IN KNUTH VOL. 2 AND
C  SEEMS TO BE QUITE RELIABLE. RAN RETURNS A VALUE UNIFORMLY SELECTED
C  BETWEEN 0 AND RANGE-1.  NOTE RESEMBLANCE TO ALG USED IN WIZARD.
C
C       IMPLICIT INTEGER(A-Z)
C       DATA R/0/
C
C       D=1
C       IF(R.NE.0)GOTO 1
C       CALL DATIME(D,T)
C       R=18*T+5
C       D=1000+MOD(D,1000)
C1      DO 2 T=1,D
C2      R=MOD(R*1021,1048576)
C       RAN=(RANGE*R)/1048576
C       RETURN
C
         INTEGER FUNCTION RAN(IRANGE)
C
C   RAN RETURNS AN INTEGER VALUE UNIFORMLY SELECTED BETWEEN
C     0 AND IRANGE -1.  SEE KNUTH, P. 155.
C   THIS ALGORITHM IS GOOD FOR MACHINES WITH WORD LENGTH OF 24 OR MORE
C
         DATA J/0/
         IF (J) 20,10,20
10       CALL TDATE(J)
20       CONTINUE
         J = AND (8388607, J * 8385709 )
         J = AND ( 8388607,J + 1772721)
         RRAN = J
         RAN = (RRAN/8388607.) * IRANGE
         RETURN
       END
