=== block 'magic item'  kind 00  270 words  (block 36)
unit    findiind(a1,wflag,oflag,search) $$ get index of item
        merge,previous
        Own(3),ncount,tcount,ocount,limit,search
          segment,own=Own(1),1  $$ ownership bitmap
*entry: a1      = number of chars
*       wflag   = 0:special, 1:weapon, 2:either
*       oflag   = 0:must not own, 1:must own, 2:either
*       search  = 0: either, 1: item slots, 2: guild lockers
*       util1-2 = item name
*       at      = location of arrow
*exit:  return  = index of the item, or -1 if no match
*
if      a1 > 10         $$ set up partial name mask
.       calc    util3« lmask(60)
.               util4« lmask(zbpc*(a1-zcpw))
else    
.       calc    util3« lmask(zbpc*a1)
.               util4« 0
endif
*
calcs   search-1,ind« 0,0,nmislot
calcs   search-1,limit« 2*nmislot,nmislot,2*nmislot
calc    Own« 0  $$ build bitmap of things I own
loop    add1(ind) {^72} limit
reloop  mislot(ind) < 1
.       calc    own(mislot(ind))« 1
endloop 
*
calc    ind« return« ncount« tcount« ocount« 0
loop    add1(ind) {^72} nmitem
*
*-- first check for partial match of name
reloop  util1 {^54} [iname(ind)$mask$util3] $$ test name for
reloop  util2 {^54} [iname2(ind)$mask$util4]$$   partial match
.       calc    add1(ncount)    $$ got a partial name
*
*-- next check other constraints
reloop  (weapon(ind)+wflag) = 1
.       calc    add1(tcount)    $$ the type is right
reloop  (own(ind)+oflag) = 1
.       calc    add1(ocount)    $$ ownership is right too
.               return« ind
*
*-- finally check for a perfect match of the name
reloop  util1 {^54} iname(ind)
reloop  util2 {^54} (iname2(ind) $mask$ lmask(42))
.       calc    ncount« ocount« tcount« 1
outloop 
endloop 
*
at      at
erase   20
if      ncount = 0      $$ not even a partial name matched
.       calc    return« -1
.       write   Not an item.
elseif  tcount = 0      $$ no matches of correct type
.       calc    return« -1
.       writec  wflag{^56}-{^56}Not a magic item.{^56}Not a weapon.{^56}+
elseif  ocount = 0      $$ no matches with right ownership
.       calc    return« -1
.       writec  oflag{^56}-{^56}Already have one.{^56}Don't own one.{^56}+
elseif  ocount = 1      $$ exactly one possibility
.       showa   iname(return),nichnam
else
.       calc    return« -1
.       write   Ambiguous name.
endif   
pause   1
at      at
erase   20
*
