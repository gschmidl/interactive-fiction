        indat(ii,aa,cc,ll) = aa + 100*ll*frac((ii-1)/ll) +    cc*int((ii-1)/ll)

        maxbags(xx)   = 5 - ceiling(xx/200000)

        select(xx,aa,bb) = bb-(aa-bb)*xx  $$if⌫⌫__ xx t⌫_ aa e⌫_ bb

        char(oo,cc)= rmask(6) $mask$ [oo $cls$ (6*cc)]

        integer:  str

          segmentf,c2=str,7,6
