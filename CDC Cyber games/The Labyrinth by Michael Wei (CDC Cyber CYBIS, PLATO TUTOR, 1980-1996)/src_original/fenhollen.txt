*  This routine used only in map editor
*  arg1 is the » location
*  arg2 is the cursor location
*  arg3 is the return string length
*  arg5 is the number of spaces to erase after »
*

at      arg1
erase   arg5

long    1

judge   ok
calcs   ktype-1,dcol«-1,0,1,1,1,0,-1,-1,-5,0,5,5,5,0,-5,-5
calcs   ktype-1,drow«-1,-1,-1,0,1,1,1,0,-5,-5,-5,0,5,5,5,0
if      ktype{^73}0
.       do      movecur
.       branch  1end
endif
long    arg3
copy    kcopya,arg3
if      key=copy
.       press   copy1
.       judge   exit
else
.       judge   key=next,rejudge,exit
endif
storea  return
ok
1end
endarrow

.       mode    rewrite
.       at      arg2
.       erase   arg3+1

.       mode    write
.       calc    escape«0
