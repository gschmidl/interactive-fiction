=== block 'run'  kind 00  222 words  (block 4)
unit    run
back    titlen
loop
.       release common
.       color   display;zyellow
.       at      1010
.       erase   50,4
.       write   Run what character
.       do      aarrow(1029,10)
reloop  return=0  $$ don't allow null reply
.       do      reserve
*
* Look up at list of currently running users (4500-6000).
* Find returns in num1 location of found signon, or -1 if
* not found. "return" is name from aarow above.
.       comload nc1,4501,1500
.       find    return,pseudin(1),30,num1
.       at      1110
.       writec  num1,,Char in use!
reloop  num1 {^54} -1  $$ if in use, abort..try again.
*
* Loop through the 3 available banks of common to
* find this guy's pseudo. If not, he ain't there. If,go on.
.       calc    ptr1 « 0
.       loop
.       .       do      comld[ add1(ptr1) ]
.       .       find    return,nc1,1500,num1,20
.       outloop num1{^73}0 $or$ ptr1{^73}3
.       endloop
.       at      1110
.       writec  num1,Char not found!,,
reloop  num1 $$  (-1 = not found)
*
* Now check the password as found in the student bank.
* Operator exempt from this check. (Nice, eh?)
.       at      1210
.       write   Type the character's password » 
.       getcode num2
outloop [ num2=nc(num1+2) ] $or$ [ op < 0 ]
.       at      1310
.       write   Wrong Password
.       release common
.       pause   keys=next,back,term
endloop
color   display;zwhite
*
* Now move his student statistics from common into student
* variables.
block   nc[ -1(num1=-1)+(num1+1) ],n(KC+1),20
comload nc1,4501,1500
do      adduser  $$ go put his name into pseudin slot..
release common
*
* Restore monsters and such.
calc    monster « savemon
        pgold   « 0
        useday  « int(day)
        actzid  « 75(ptr1)-int[ num1/20 ]
do      findmnum
*do      clean
jump    dungeon
**

unit    writeit

* Save "who usedit" data. not really necessary..but..

.       pack    savgrp(savptr),,«t,abs(tips),2»«a,zgroup,8»

comload nc1,1,1500

* If the player was alive (phits>0)
if      {plev>[(nc(7)$ars$11)$mask$rmask(7)]} $and$ phits>0
.       comload nc1,4501,1500
*
* If the player was in game, then turn off his
* "I am alive" stats in common.
.       if      player>0
.       .       calc    sub1m(numuser,0)
.       .               pseudin(player)«goldin(player)«0
.       .               gotmess(player)«0
.       .               player«0
.       endif
*
* grab common. Can exec error error here if takes too long.
.       loop
.       .       reserve common
.       outloop zreturn
.       endloop
*
* Loop through the three sections of common,1-1500,1501-3000
* and 3001-4500. There player statistics are stored. Hence
* There's 1500/20, or 75 x 3 = 225 available slots.
.       calc    ptr1 « 0
*
* Go find his pseudo in common..bank 1,2, then 3(comld)
.       loop
.       .       do      comld[ add1(ptr1) ]
.       .       find    pseudo,nc1,1500,num1,20
.       outloop num1{^73}0 $or$ ptr1{^73}3
.       endloop
*
* Save any monsters currently with him..*grin*
.       calc    savemon«monster
.               savelev«monlev
.               savenum«monnum
*
* And block move his player stats, up in high student vars,
* to common.
.       block   n(KC+1),nc[ -1(num1=-1)+(num1+1) ],20
.       comload
.       sort    common,1;225,20,403,7
else
.       do      subuser
endif
release common
