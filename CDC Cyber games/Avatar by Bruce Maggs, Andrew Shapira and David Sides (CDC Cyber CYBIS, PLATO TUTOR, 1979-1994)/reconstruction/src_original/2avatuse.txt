unit    itemck(litem,lfrom)
        merge,previous
        litem           $$ item slot (1-slots)
        linuse          $$ item in use of iclass

        lfrom           $$ -1 = P, 0 = U, 1 = drop

        labil           $$ ability level (temp)

*--

        lusing « (linuse = litem)
        lunuse « lusing $mask$ (lfrom < 1)
        lstuck « istuck(iclass)
        return « 2
        branch  lstuck $mask$ (lfrom > -1),1bad,x
        branch  (eq1(lfrom)) $union$ lunuse,1ok,x
        return « 0
        branch  (eq1(chibank(litem)))$mask$(zloc > 0),1bad,x
        return « 5
        branch  iclassu(class)-1,1bad,x
        lstat « 0
loop    (lstat « lstat+1) {^72} 6
reloop  istat(lstat) {^72} chstat(lstat)
.       calc    return « 5 + lstat
.               branch  1bad

calc    return « 3
        branch  (twohand $mask$ (iclass = 4)),1bad,x
        return « 4
        branch  ((chitemu(4) > 0) $mask$ (eq1(ispecb(i.twoh)))),1bad,x
if      (labil « iabil) > 0

.               branch  (pabilit(labil)-ialev),1bad,x
